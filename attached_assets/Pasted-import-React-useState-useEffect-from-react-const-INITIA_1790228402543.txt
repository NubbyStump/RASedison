import React, { useState, useEffect } from 'react';

const INITIAL_GROUPS = [
  { id: 'ladybugs', name: 'Ladybugs', color: 'from-pink-500 to-rose-600', score: 0 },
  { id: 'jellyfish', name: 'Jellyfish', color: 'from-cyan-500 to-blue-600', score: 0 },
  { id: 'tigers', name: 'Tigers', color: 'from-amber-500 to-orange-600', score: 0 }
];

const PRELOADED_SCRAMBLES = [
  {
    id: 's1',
    title: 'THE SLIME MATRIX',
    points: 600,
    scrambled: 'C O O E P R T A I N O   I S   Y E K',
    solution: 'COOPERATION IS KEY',
    hint: 'Working together as a team unlocks the Slime Box!',
    type: 'whiteboard'
  },
  {
    id: 's2',
    title: 'AFTER SCHOOL RULE #1',
    points: 400,
    scrambled: 'R E S E P C T   E V E R Y O N E',
    solution: 'RESPECT EVERYONE',
    hint: 'How we treat our teammates and counselors.',
    type: 'whiteboard'
  },
  {
    id: 's3',
    title: 'CLEANUP CHAMPIONS',
    points: 300,
    scrambled: 'L E A V E   N O   T R A C E',
    solution: 'LEAVE NO TRACE',
    hint: 'What we do when free play finishes.',
    type: 'whiteboard'
  }
];

export default function App() {
  const [activeTab, setActiveTab] = useState('points');
  const [groups, setGroups] = useState(() => {
    const saved = localStorage.getItem('ras_groups');
    return saved ? JSON.parse(saved) : INITIAL_GROUPS;
  });
  const [reason, setReason] = useState('');
  const [logs, setLogs] = useState(() => {
    const saved = localStorage.getItem('ras_logs');
    return saved ? JSON.parse(saved) : [];
  });
  
  // Fastest Lap State
  const [lapTimes, setLapTimes] = useState(() => {
    const saved = localStorage.getItem('ras_laps');
    return saved ? JSON.parse(saved) : [];
  });
  const [runnerName, setRunnerName] = useState('');
  const [selectedGroup, setSelectedGroup] = useState('ladybugs');
  const [lapMinutes, setLapMinutes] = useState('0');
  const [lapSeconds, setLapSeconds] = useState('00');
  const [lapMillis, setLapMillis] = useState('00');

  // Days left calculation
  const now = new Date();
  const lastDay = new Date(now.getFullYear(), now.getMonth() + 1, 0).getDate();
  const daysLeft = lastDay - now.getDate();

  useEffect(() => {
    localStorage.setItem('ras_groups', JSON.stringify(groups));
  }, [groups]);

  useEffect(() => {
    localStorage.setItem('ras_logs', JSON.stringify(logs));
  }, [logs]);

  useEffect(() => {
    localStorage.setItem('ras_laps', JSON.stringify(lapTimes));
  }, [lapTimes]);

  const maxScore = Math.max(...groups.map(g => g.score));
  const leadingGroups = groups.filter(g => g.score === maxScore && g.score > 0);

  const addPoints = (groupId, amount) => {
    const updated = groups.map(g => g.id === groupId ? { ...g, score: g.score + amount } : g);
    setGroups(updated);
    
    const targetGroup = groups.find(g => g.id === groupId);
    const logEntry = {
      id: Date.now(),
      groupName: targetGroup.name,
      groupId,
      amount,
      reason: reason || 'Group Effort',
      time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
    };
    setLogs([logEntry, ...logs]);
    setReason('');
  };

  const undoLog = (logId) => {
    const targetLog = logs.find(l => l.id === logId);
    if (!targetLog) return;
    setGroups(groups.map(g => g.id === targetLog.groupId ? { ...g, score: g.score - targetLog.amount } : g));
    setLogs(logs.filter(l => l.id !== logId));
  };

  const saveLapTime = (e) => {
    e.preventDefault();
    if (!runnerName) return;
    const totalMs = (parseInt(lapMinutes || '0') * 60000) + (parseInt(lapSeconds || '0') * 1000) + (parseInt(lapMillis || '0') * 10);
    const newEntry = {
      id: Date.now(),
      runnerName,
      group: selectedGroup,
      timeFormatted: `${lapMinutes.padStart(2, '0')}:${lapSeconds.padStart(2, '0')}.${lapMillis.padStart(2, '0')}`,
      totalMs,
      date: new Date().toLocaleDateString()
    };
    setLapTimes([...lapTimes, newEntry].sort((a, b) => a.totalMs - b.totalMs));
    setRunnerName('');
  };

  const fastestLapHolder = lapTimes[0];

  return (
    <div className="min-h-screen bg-slate-900 text-slate-100 font-sans pb-12">
      {/* App Header */}
      <header className="bg-slate-800 border-b border-slate-700 p-4 sticky top-0 z-50">
        <div className="max-w-4xl mx-auto flex flex-col sm:flex-row items-center justify-between gap-2">
          <div className="text-center sm:text-left">
            <h1 className="text-xl font-bold text-amber-400 tracking-wide flex items-center justify-center sm:justify-start gap-2">
              ✨ Right At School
            </h1>
            <p className="text-xs text-slate-400 font-medium">Edison Language Academy</p>
          </div>
          
          {/* Navigation Bar */}
          <nav className="flex bg-slate-900 p-1 rounded-xl border border-slate-700 gap-1 text-xs">
            <button 
              onClick={() => setActiveTab('points')}
              className={`px-3 py-1.5 rounded-lg font-bold transition ${activeTab === 'points' ? 'bg-amber-500 text-slate-950' : 'text-slate-400 hover:text-white'}`}
            >
              Points
            </button>
            <button 
              onClick={() => setActiveTab('scrambles')}
              className={`px-3 py-1.5 rounded-lg font-bold transition ${activeTab === 'scrambles' ? 'bg-amber-500 text-slate-950' : 'text-slate-400 hover:text-white'}`}
            >
              Scrambles
            </button>
            <button 
              onClick={() => setActiveTab('laps')}
              className={`px-3 py-1.5 rounded-lg font-bold transition ${activeTab === 'laps' ? 'bg-amber-500 text-slate-950' : 'text-slate-400 hover:text-white'}`}
            >
              Fastest Lap
            </button>
            <button 
              onClick={() => setActiveTab('projector')}
              className={`px-3 py-1.5 rounded-lg font-bold transition flex items-center gap-1 ${activeTab === 'projector' ? 'bg-lime-500 text-slate-950' : 'text-lime-400 hover:text-lime-300'}`}
            >
              📺 TV Mode
            </button>
          </nav>
        </div>
      </header>

      {/* Main Container */}
      <main className="max-w-4xl mx-auto p-4">
        {/* POINTS TAB */}
        {activeTab === 'points' && (
          <div className="space-y-6">
            {/* Status Header */}
            <div className="bg-slate-800/80 border border-slate-700 rounded-2xl p-4 text-center">
              <span className="text-xs uppercase tracking-wider text-slate-400 font-semibold">Monthly Race Status</span>
              <h2 className="text-lg font-black text-amber-400 mt-1">
                {leadingGroups.length > 0 
                  ? `${leadingGroups.map(g => g.name).join(' & ')} Leading by ${maxScore} pts!`
                  : 'Game On! Add points to take the lead!'}
              </h2>
              <p className="text-xs text-lime-400 font-bold mt-1">
                🧪 {daysLeft} Days Remaining to Win Slime & Prizes!
              </p>
            </div>

            {/* Optional Reason Input */}
            <div className="bg-slate-800 p-3 rounded-xl border border-slate-700">
              <input 
                type="text" 
                placeholder="Reason for points (e.g., Quietest Line, Super Scramble Win)..."
                value={reason}
                onChange={(e) => setReason(e.target.value)}
                className="w-full bg-slate-900 border border-slate-700 rounded-lg px-3 py-2 text-sm text-slate-100 placeholder-slate-500 focus:outline-none focus:border-amber-500"
              />
            </div>

            {/* Group Cards */}
            <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
              {groups.map((group) => {
                const isLeading = group.score === maxScore && group.score > 0;
                return (
                  <div 
                    key={group.id} 
                    className={`bg-slate-800 rounded-2xl border ${isLeading ? 'border-lime-500 shadow-lg shadow-lime-500/10' : 'border-slate-700'} p-4 relative overflow-hidden flex flex-col justify-between`}
                  >
                    {/* Slime Drip Accent for Leaders */}
                    {isLeading && (
                      <div className="absolute top-0 left-0 right-0 bg-lime-500 text-slate-950 font-black text-[10px] text-center py-0.5 tracking-wider uppercase">
                        🟢 SLIME LEADER 🟢
                      </div>
                    )}

                    <div className="mt-2 text-center">
                      <h3 className="text-2xl font-black text-slate-100">{group.name}</h3>
                      <div className="text-5xl font-black my-3 text-amber-400">{group.score}</div>
                    </div>

                    {/* Point Add Buttons */}
                    <div className="grid grid-cols-3 gap-1.5 mt-2">
                      <button onClick={() => addPoints(group.id, 10)} className="bg-slate-700 hover:bg-slate-600 text-slate-100 font-bold py-2 rounded-lg text-sm transition">+10</button>
                      <button onClick={() => addPoints(group.id, 50)} className="bg-slate-700 hover:bg-slate-600 text-slate-100 font-bold py-2 rounded-lg text-sm transition">+50</button>
                      <button onClick={() => addPoints(group.id, 100)} className="bg-slate-700 hover:bg-slate-600 text-slate-100 font-bold py-2 rounded-lg text-sm transition">+100</button>
                      <button onClick={() => addPoints(group.id, 200)} className="bg-amber-500 hover:bg-amber-400 text-slate-950 font-extrabold py-2 rounded-lg text-sm transition">+200</button>
                      <button onClick={() => addPoints(group.id, 600)} className="bg-lime-500 hover:bg-lime-400 text-slate-950 font-extrabold py-2 rounded-lg text-sm transition col-span-2">
                        +600 Scramble
                      </button>
                    </div>
                  </div>
                );
              })}
            </div>

            {/* Recent Point Activity Log */}
            <div className="bg-slate-800 p-4 rounded-2xl border border-slate-700">
              <h3 className="text-sm font-bold text-slate-300 mb-3">Recent Points History</h3>
              {logs.length === 0 ? (
                <p className="text-xs text-slate-500 italic">No points awarded yet today.</p>
              ) : (
                <div className="space-y-2 max-h-48 overflow-y-auto pr-1">
                  {logs.map((log) => (
                    <div key={log.id} className="flex items-center justify-between bg-slate-900/80 p-2.5 rounded-lg border border-slate-800 text-xs">
                      <div>
                        <span className="font-bold text-amber-400">{log.groupName}</span>
                        <span className="text-slate-400 ml-2">({log.reason})</span>
                      </div>
                      <div className="flex items-center gap-3">
                        <span className="font-extrabold text-lime-400">+{log.amount}</span>
                        <button onClick={() => undoLog(log.id)} className="text-slate-500 hover:text-rose-400 font-bold">
                          ↺
                        </button>
                      </div>
                    </div>
                  ))}
                </div>
              )}
            </div>
          </div>
        )}

        {/* WHITEBOARD SCRAMBLES TAB */}
        {activeTab === 'scrambles' && (
          <div className="space-y-6">
            <div className="bg-slate-800 p-4 rounded-2xl border border-slate-700">
              <h2 className="text-lg font-bold text-amber-400 flex items-center gap-2">
                ⚡ Whiteboard Super Scrambles
              </h2>
              <p className="text-xs text-slate-400 mt-1">
                Write these on the classroom whiteboard. First group to decipher the phrase gets 200-600 points!
              </p>
            </div>

            <div className="grid grid-cols-1 gap-4">
              {PRELOADED_SCRAMBLES.map((scramble) => (
                <div key={scramble.id} className="bg-slate-800 rounded-2xl border border-slate-700 p-5 space-y-3">
                  <div className="flex items-center justify-between">
                    <span className="bg-amber-500/10 text-amber-400 border border-amber-500/30 text-xs font-black px-2.5 py-1 rounded-full">
                      {scramble.points} POINTS
                    </span>
                    <span className="text-xs font-bold text-slate-400 uppercase tracking-wider">{scramble.title}</span>
                  </div>

                  <div className="bg-slate-950 p-4 rounded-xl border border-slate-800 text-center font-mono tracking-widest text-lg font-bold text-lime-400 my-2">
                    {scramble.scrambled}
                  </div>

                  <div className="text-xs space-y-1">
                    <p className="text-slate-300 font-semibold">💡 Hint: <span className="text-slate-400 font-normal">{scramble.hint}</span></p>
                    <p className="text-slate-300 font-semibold">✅ Solution: <span className="text-lime-400 font-mono font-bold">{scramble.solution}</span></p>
                  </div>
                </div>
              ))}
            </div>
          </div>
        )}

        {/* FASTEST LAP TAB */}
        {activeTab === 'laps' && (
          <div className="space-y-6">
            {/* Record Form */}
            <div className="bg-slate-800 p-4 rounded-2xl border border-slate-700">
              <h2 className="text-lg font-bold text-amber-400 mb-3 flex items-center gap-2">
                ⏱️ Record a Lap Time
              </h2>
              <form onSubmit={saveLapTime} className="space-y-3 text-xs">
                <div>
                  <label className="block text-slate-400 mb-1">Student Name or Group</label>
                  <input 
                    type="text" 
                    value={runnerName}
                    onChange={(e) => setRunnerName(e.target.value)}
                    placeholder="e.g., Alex S. or Tigers Relay Team"
                    className="w-full bg-slate-900 border border-slate-700 rounded-lg px-3 py-2 text-slate-100 focus:outline-none focus:border-amber-500"
                  />
                </div>

                <div>
                  <label className="block text-slate-400 mb-1">Group</label>
                  <select 
                    value={selectedGroup}
                    onChange={(e) => setSelectedGroup(e.target.value)}
                    className="w-full bg-slate-900 border border-slate-700 rounded-lg px-3 py-2 text-slate-100 focus:outline-none"
                  >
                    {groups.map(g => <option key={g.id} value={g.id}>{g.name}</option>)}
                  </select>
                </div>

                <div className="grid grid-cols-3 gap-2">
                  <div>
                    <label className="block text-slate-400 mb-1">Minutes</label>
                    <input type="number" min="0" value={lapMinutes} onChange={(e) => setLapMinutes(e.target.value)} className="w-full bg-slate-900 border border-slate-700 rounded-lg p-2 text-center text-slate-100" />
                  </div>
                  <div>
                    <label className="block text-slate-400 mb-1">Seconds</label>
                    <input type="number" min="0" max="59" value={lapSeconds} onChange={(e) => setLapSeconds(e.target.value)} className="w-full bg-slate-900 border border-slate-700 rounded-lg p-2 text-center text-slate-100" />
                  </div>
                  <div>
                    <label className="block text-slate-400 mb-1">Millis</label>
                    <input type="number" min="0" max="99" value={lapMillis} onChange={(e) => setLapMillis(e.target.value)} className="w-full bg-slate-900 border border-slate-700 rounded-lg p-2 text-center text-slate-100" />
                  </div>
                </div>

                <button type="submit" className="w-full bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold py-2.5 rounded-lg transition mt-2">
                  Save Lap Time
                </button>
              </form>
            </div>

            {/* Leaderboard */}
            <div className="bg-slate-800 p-4 rounded-2xl border border-slate-700">
              <h3 className="text-sm font-bold text-slate-300 mb-3">Month's Leaderboard</h3>
              {lapTimes.length === 0 ? (
                <p className="text-xs text-slate-500 italic">No laps logged yet this month.</p>
              ) : (
                <div className="space-y-2">
                  {lapTimes.map((lap, index) => (
                    <div key={lap.id} className={`flex items-center justify-between p-3 rounded-xl border text-xs ${index === 0 ? 'bg-amber-500/10 border-amber-500/40' : 'bg-slate-900 border-slate-800'}`}>
                      <div className="flex items-center gap-3">
                        <span className={`font-black w-5 text-center ${index === 0 ? 'text-amber-400 text-sm' : 'text-slate-500'}`}>
                          {index === 0 ? '👑' : `#${index + 1}`}
                        </span>
                        <div>
                          <p className="font-bold text-slate-100">{lap.runnerName}</p>
                          <p className="text-[10px] text-slate-400 uppercase">{lap.group}</p>
                        </div>
                      </div>
                      <span className="font-mono font-bold text-lime-400 text-sm">{lap.timeFormatted}</span>
                    </div>
                  ))}
                </div>
              )}
            </div>
          </div>
        )}

        {/* TV / PROJECTOR MODE */}
        {activeTab === 'projector' && (
          <div className="min-h-[80vh] bg-slate-950 rounded-3xl border-2 border-slate-800 p-6 flex flex-col justify-between text-center">
            <div>
              <h1 className="text-2xl sm:text-4xl font-black text-amber-400 tracking-wider uppercase">
                Right At School
              </h1>
              <p className="text-slate-400 font-bold text-sm sm:text-lg mt-1">Edison Language Academy Leaderboard</p>
              <p className="text-lime-400 font-bold text-xs sm:text-sm mt-2">🧪 {daysLeft} Days Remaining to Win Slime!</p>
            </div>

            {/* Giant Scorecards */}
            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 my-8">
              {groups.map((group) => {
                const isLeading = group.score === maxScore && group.score > 0;
                return (
                  <div key={group.id} className={`bg-slate-900 rounded-2xl p-6 border-2 ${isLeading ? 'border-lime-400 shadow-2xl shadow-lime-500/20 scale-105' : 'border-slate-800'} relative`}>
                    {isLeading && (
                      <div className="absolute -top-3 left-1/2 -translate-x-1/2 bg-lime-400 text-slate-950 font-black text-xs px-3 py-1 rounded-full uppercase">
                        👑 LEADER
                      </div>
                    )}
                    <h2 className="text-xl font-bold text-slate-200">{group.name}</h2>
                    <div className="text-6xl sm:text-7xl font-black text-amber-400 my-4">{group.score}</div>
                  </div>
                );
              })}
            </div>

            {/* Fastest Lap Highlight */}
            {fastestLapHolder && (
              <div className="bg-slate-900 border border-slate-800 rounded-xl p-3 max-w-md mx-auto text-xs">
                <span className="text-slate-400 uppercase font-bold tracking-wider">⚡ Month's Fastest Lap Holder</span>
                <p className="text-amber-400 font-bold text-sm mt-0.5">{fastestLapHolder.runnerName} — <span className="text-lime-400 font-mono">{fastestLapHolder.timeFormatted}</span></p>
              </div>
            )}
          </div>
        )}
      </main>
    </div>
  );
}
